# ==============================================================================
# Batch-Processing Data Architecture
# Docker Compose Microservices Orchestration
# ==============================================================================

version: '3.8'

networks:
  data-net:
    driver: bridge

volumes:
  postgres_data:
  minio_data:

services:
  # 1. Relational Metadata & Feature Store (PostgreSQL)
  postgres:
    image: postgres:15-alpine
    container_name: de_postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: feature_store
      POSTGRES_USER: db_user
      POSTGRES_PASSWORD: db_password
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./sql/schema.sql:/docker-entrypoint-initdb.d/schema.sql
    networks:
      - data-net
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U db_user -d feature_store"]
      interval: 5s
      timeout: 5s
      retries: 5

  # 2. Local S3-Compatible Object Store (MinIO)
  minio:
    image: minio/minio:RELEASE.2023-09-20T22-40-07Z
    container_name: de_minio
    restart: unless-stopped
    command: server /data --console-address ":9001"
    environment:
      MINIO_ROOT_USER: minio_admin
      MINIO_ROOT_PASSWORD: minio_password
    ports:
      - "9000:9000"
      - "9001:9001"
    volumes:
      - minio_data:/data
    networks:
      - data-net
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:9000/minio/health/live"]
      interval: 5s
      timeout: 5s
      retries: 5

  # MinIO Bucket Initialization Helper
  createbuckets:
    image: minio/mc:RELEASE.2023-09-20T21-38-30Z
    container_name: de_minio_init
    depends_on:
      minio:
        condition: service_healthy
    networks:
      - data-net
    entrypoint: >
      /bin/sh -c "
      /usr/bin/mc alias set myminio http://minio:9000 minio_admin minio_password;
      /usr/bin/mc mb myminio/raw-data --ignore-existing;
      /usr/bin/mc anonymous set download myminio/raw-data;
      exit 0;
      "

  # 3. Data Ingestion Microservice (Python ETL)
  ingestion:
    build:
      context: ./services/ingestion
    container_name: de_ingestion
    depends_on:
      postgres:
        condition: service_healthy
      createbuckets:
        condition: service_completed_successfully
    environment:
      - MINIO_ENDPOINT=minio:9000
      - MINIO_ACCESS_KEY=minio_admin
      - MINIO_SECRET_KEY=minio_password
      - POSTGRES_HOST=postgres
      - POSTGRES_DB=feature_store
      - POSTGRES_USER=db_user
      - POSTGRES_PASSWORD=db_password
    networks:
      - data-net

  # 4. Batch Processing Engine (Apache Spark / PySpark)
  spark-processor:
    build:
      context: ./services/spark_processor
    container_name: de_spark_processor
    depends_on:
      ingestion:
        condition: service_completed_successfully
      postgres:
        condition: service_healthy
    environment:
      - MINIO_ENDPOINT=http://minio:9000
      - MINIO_ACCESS_KEY=minio_admin
      - MINIO_SECRET_KEY=minio_password
      - POSTGRES_HOST=postgres
      - POSTGRES_DB=feature_store
      - POSTGRES_USER=db_user
      - POSTGRES_PASSWORD=db_password
    networks:
      - data-net

  # 5. Feature Delivery REST API (FastAPI)
  api:
    build:
      context: ./services/api
    container_name: de_fastapi
    restart: unless-stopped
    depends_on:
      spark-processor:
        condition: service_completed_successfully
      postgres:
        condition: service_healthy
    ports:
      - "8000:8000"
    environment:
      - POSTGRES_HOST=postgres
      - POSTGRES_DB=feature_store
      - POSTGRES_USER=db_user
      - POSTGRES_PASSWORD=db_password
    networks:
      - data-net
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/health"]
      interval: 10s
      timeout: 5s
      retries: 3
